Column

Daily Max vs. Min Temperature (°C), NYC stations, 2010

nyc_weather_df |>
  filter(!is.na(tmax), !is.na(tmin)) |>
  mutate(
    `Min Temp (°C)` = tmin,
    `Max Temp (°C)` = tmax,
    text_label = str_c(
      "Station: ", station,
      "\nDate: ", year, "-", month, "-", day,
      "\nMax temp: ", tmax, " °C",
      "\nMin temp: ", tmin, " °C")) |>
  plot_ly(
    x = ~`Min Temp (°C)`, y = ~`Max Temp (°C)`,
    type = "scatter", mode = "markers",
    color = ~station, text = ~text_label, alpha = 0.5)

Distribution of Daily Max Temperature (°C) by Month (1 = January), 2010

nyc_weather_df |>
  filter(!is.na(tmax)) |>
  mutate(
    month = factor(month),
    `Max Temp (°C)` = tmax) |>
  plot_ly(
    x = ~month, y = ~`Max Temp (°C)`, color = ~month,
    type = "box", colors = "viridis")
